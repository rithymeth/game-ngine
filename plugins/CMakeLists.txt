# The engine's own plugins (Phase 26 step 1
# §26.1): each optional module as a plugin with an .aplugin descriptor and a
# runtime module, built when the module itself is.
set(AETHER_ENGINE_PLUGINS Physics Audio Navigation AI Networking Animation UI VFX Sprite2D Sequencer Streaming Gameplay Inventory Interaction Quests)
set(_target_Physics Aether::Physics)
set(_target_Audio Aether::Audio)
set(_target_Navigation Aether::Nav)
set(_target_AI Aether::AI)
set(_target_Networking Aether::Net)
set(_target_Animation Aether::Animation)
set(_target_UI Aether::UI)
set(_target_VFX Aether::VFX)
set(_target_Sprite2D Aether::Sprite2D)
set(_target_Sequencer Aether::Sequencer)
set(_target_Streaming Aether::Streaming)
set(_target_Gameplay Aether::Gameplay)
set(_target_Inventory Aether::Inventory)
set(_target_Interaction Aether::Interaction)
set(_target_Quests Aether::Quests)
foreach(plugin ${AETHER_ENGINE_PLUGINS})
    if(TARGET ${_target_${plugin}})
        add_subdirectory(${plugin})
    endif()
endforeach()
